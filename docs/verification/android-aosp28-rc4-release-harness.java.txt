package studio.ax.arrival.journeytest;
import android.app.Instrumentation;

import org.json.JSONObject;

/** Drives public accessibility UI of a separately installed, unchanged release APK. */
public final class JourneyInstrumentation extends Instrumentation {
 private UiAutomation ui; private File out; private long started; private String runId, remoteVideo;
 private final JSONArray events=new JSONArray(); private final JSONObject result=new JSONObject();
 private final String notice="학술대회 안내\n장소: 코엑스 아셈볼룸 2층\n행사는 10:00에 시작하여 18:00에 종료됩니다.\n참가 등록은 1층 로비에서 진행합니다.";
 @Override public void onCreate(Bundle args){super.onCreate(args);runId=args.getString("run_id","");if(!runId.matches("[0-9a-f]{32}"))throw new IllegalArgumentException("Fresh run_id required");remoteVideo="/sdcard/arrival-rc4-release-scenario-"+runId+".mp4";start();}
 private int shellSequence=0;
 private String shell(String command) throws Exception {
  // UiAutomation executes an argv-style command, not shell syntax. Put all
  // quoting, redirections and background jobs in a UTF-8 script first.
  File script=new File(out,"journey-command-"+(shellSequence++)+".sh");
  try(FileOutputStream file=new FileOutputStream(script)){file.write(("#!/system/bin/sh\n"+command+"\n").getBytes("UTF-8"));}
  try(ParcelFileDescriptor fd=ui.executeShellCommand("/system/bin/sh "+script.getAbsolutePath());InputStream in=new ParcelFileDescriptor.AutoCloseInputStream(fd);ByteArrayOutputStream b=new ByteArrayOutputStream()){
   byte[] buf=new byte[8192];int n;while((n=in.read(buf))!=-1)b.write(buf,0,n);return b.toString("UTF-8");
  }finally{script.delete();}
 }
 private String quote(String s){return "'"+s.replace("'","'\\''")+"'";}
 private void waitUi(){SystemClock.sleep(220);try{ui.waitForIdle(120,1800);}catch(Exception ignored){}}
 private AccessibilityNodeInfo find(AccessibilityNodeInfo node,String text,boolean desc,boolean contains){
  if(node==null)return null;CharSequence cs=desc?node.getContentDescription():node.getText();String val=cs==null?"":cs.toString();
  if(node.isVisibleToUser()&&(contains?val.contains(text):val.equals(text)))return node;
  for(int i=0;i<node.getChildCount();i++){AccessibilityNodeInfo hit=find(node.getChild(i),text,desc,contains);if(hit!=null)return hit;}return null;
 }
 private AccessibilityNodeInfo scrollNode(AccessibilityNodeInfo n){if(n==null)return null;if(n.isScrollable()&&n.isVisibleToUser())return n;for(int i=0;i<n.getChildCount();i++){AccessibilityNodeInfo hit=scrollNode(n.getChild(i));if(hit!=null)return hit;}return null;}
 private AccessibilityNodeInfo locate(String text,boolean desc,boolean contains,boolean up) throws Exception {
  for(int attempt=0;attempt<9;attempt++){
   waitUi();AccessibilityNodeInfo root=ui.getRootInActiveWindow();AccessibilityNodeInfo hit=find(root,text,desc,contains);if(hit!=null)return hit;
   AccessibilityNodeInfo scroller=scrollNode(root);if(scroller!=null)scroller.performAction(up?AccessibilityNodeInfo.ACTION_SCROLL_BACKWARD:AccessibilityNodeInfo.ACTION_SCROLL_FORWARD);
   // ACTION_CLICK is asynchronous. A dialog with no scroller may still be
   // closing while the next target is absent; retry the fresh active root.
   SystemClock.sleep(450);
  }throw new AssertionError("Visible target absent: "+text);
 }
 private void tap(String text) throws Exception {tap(text,false,false);}
 private void tap(String text,boolean contains,boolean up) throws Exception {AccessibilityNodeInfo n=locate(text,false,contains,up);if(!n.isEnabled())throw new AssertionError("Disabled target: "+text);while(n!=null&&!n.isClickable())n=n.getParent();if(n==null||!n.performAction(AccessibilityNodeInfo.ACTION_CLICK))throw new AssertionError("Click failed: "+text);waitUi();}
 private void enterDemoAndAwaitSharedCheck() throws Exception {
  tap("차량 밖에서 데모 시작");tap("데모 시작");
  long start=SystemClock.elapsedRealtime(),deadline=start+15000;boolean touchFallback=false;
  while(SystemClock.elapsedRealtime()<deadline){
   AccessibilityNodeInfo root=ui.getRootInActiveWindow();
   if(find(root,"공유한 안내를 가져오기 전, 안전하게 주차했어요.",false,false)!=null){result.put("demo_entry_public_touch_fallback",touchFallback);return;}
   // ACTION_CLICK may acknowledge before a slow dialog transition. If the
   // exact public dialog button is still visible, one touch uses its fresh bounds.
   if(!touchFallback&&SystemClock.elapsedRealtime()-start>=5000){
    AccessibilityNodeInfo button=find(root,"데모 시작",false,false);
    if(button!=null&&button.isEnabled()){
     Rect bounds=new Rect();button.getBoundsInScreen(bounds);if(bounds.isEmpty())throw new AssertionError("Demo dialog button has no visible bounds");
     long down=SystemClock.uptimeMillis();MotionEvent press=MotionEvent.obtain(down,down,MotionEvent.ACTION_DOWN,bounds.exactCenterX(),bounds.exactCenterY(),0);press.setSource(InputDevice.SOURCE_TOUCHSCREEN);
     boolean pressed=ui.injectInputEvent(press,true);press.recycle();SystemClock.sleep(70);
     MotionEvent release=MotionEvent.obtain(down,SystemClock.uptimeMillis(),MotionEvent.ACTION_UP,bounds.exactCenterX(),bounds.exactCenterY(),0);release.setSource(InputDevice.SOURCE_TOUCHSCREEN);
     boolean released=ui.injectInputEvent(release,true);release.recycle();if(!pressed||!released)throw new AssertionError("Public demo dialog touch failed");touchFallback=true;
    }
   }
   SystemClock.sleep(300);
  }
  throw new AssertionError("Demo confirmation did not expose shared parking check within 15 seconds");
 }
 private void set(String desc,String value) throws Exception {AccessibilityNodeInfo n=locate(desc,true,false,false);Bundle b=new Bundle();b.putCharSequence(AccessibilityNodeInfo.ACTION_ARGUMENT_SET_TEXT_CHARSEQUENCE,value);if(!n.isEnabled()||!n.performAction(AccessibilityNodeInfo.ACTION_SET_TEXT,b))throw new AssertionError("Set text failed: "+desc);waitUi();}
 private void check(String text,boolean contains) throws Exception {AccessibilityNodeInfo n=locate(text,false,contains,false);if(!n.isChecked()){if(!n.isEnabled()||!n.performAction(AccessibilityNodeInfo.ACTION_CLICK))throw new AssertionError("Check failed:"+text);waitUi();}}
 private void event(String label) throws Exception {JSONObject e=new JSONObject();e.put("label",label);e.put("seconds_from_scenario_start",(SystemClock.elapsedRealtime()-started)/1000.0);events.put(e);Bundle b=new Bundle();b.putString("stream",label+"\n");sendStatus(1,b);}
 private void capture(String label) throws Exception {Bitmap b=ui.takeScreenshot();if(b==null)throw new AssertionError("Screenshot absent");try(FileOutputStream f=new FileOutputStream(new File(out,"release-"+label+".png"))){b.compress(Bitmap.CompressFormat.PNG,100,f);}b.recycle();event(label);}
 private String allText(AccessibilityNodeInfo n){if(n==null||!n.isVisibleToUser())return "";StringBuilder b=new StringBuilder();if(n.getText()!=null)b.append(n.getText()).append('\n');for(int i=0;i<n.getChildCount();i++)b.append(allText(n.getChild(i)));return b.toString();}

 private void assertQrAndCapture(String label) throws Exception {
  AccessibilityNodeInfo qr=locate("저장한 주차 위치를 휴대폰으로 가져가는 QR",true,false,false);
  Rect r=new Rect();qr.getBoundsInScreen(r);
  if(r.left<0||r.right>1280||r.top<24||r.bottom>752||r.width()<256||r.width()!=r.height())throw new AssertionError("QR not fully visible and square: "+r);
  JSONObject bound=new JSONObject();bound.put("left",r.left);bound.put("top",r.top);bound.put("right",r.right);bound.put("bottom",r.bottom);result.put(label+"_bounds",bound);
  capture(label);
 }
 private void launchForFont(String scale) throws Exception {
  shell("settings put system font_scale "+scale);
  String actual=shell("settings get system font_scale").trim();if(!scale.equals(actual))throw new AssertionError("Font setting mismatch: "+actual);
  SystemClock.sleep(1800);
  shell("am force-stop studio.ax.arrival");
  SystemClock.sleep(350);
  String launch=shell("am start -W -a android.intent.action.SEND -t text/plain --es android.intent.extra.TEXT "+quote(notice)+" -n studio.ax.arrival/.MainActivity");
  result.put("launch_response_font_"+scale,launch);
  if(!launch.contains("Status: ok"))throw new AssertionError("Launch failed; response="+launch);
  waitUi();enterDemoAndAwaitSharedCheck();check("공유한 안내를 가져오기 전, 안전하게 주차했어요.",false);tap("공유한 행사 안내 가져오기");
  if(!notice.contentEquals(locate("행사 장소 안내",true,false,false).getText()))throw new AssertionError("Original notice changed");
 }
 @Override public void onStart(){boolean passed=false;try{
  ui=getUiAutomation();out=getContext().getExternalFilesDir(null);out.mkdirs();
  for(File prior:out.listFiles())if(prior.isFile()&&(prior.getName().startsWith("release-")||prior.getName().equals("release-scenario.json"))&&!prior.delete())throw new IOException("Cannot isolate prior artifact");
  started=SystemClock.elapsedRealtime();result.put("run_id",runId);result.put("version","1.0.0-rc4");result.put("build_type","release");result.put("automation","Public accessibility UI actions from a separate self-targeted test harness; no app-private API or app instrumentation.");result.put("scope","Screenshots only; no recording and no performance measurement");result.put("notice",notice);result.put("memo","B3 블루존 E12");
  launchForFont("1.0");
  String helper="개인정보만 지우고, 장소·일정·등록·변경·취소·진입·주차 제한 안내는 원문대로 남겨주세요.";
  locate(helper,false,false,true);capture("input-helper");result.put("input_helper",helper);
  set("루프박스와 적재물을 포함한 차량 전체 높이, 미터","1.65");check("적재물을 포함한 전체 높이를 확인했습니다.",false);check("현재 유효한 행사의 실제 방문 장소",true);check("안전하게 주차한 상태입니다.",false);tap("도착 브리핑 확정");
  String confirmed=allText(ui.getRootInActiveWindow());if(!confirmed.contains("아셈볼룸")||!confirmed.contains("북문")||!confirmed.contains("B3 블루존"))throw new AssertionError("Confirmed pass incomplete");capture("confirmed");
  tap("주차 위치 기록");check("차량을 안전하게 주차했고, 위치 메모를 확인할게요.",false);set("직접 적는 주차 위치 메모, 최대 64자","B3 블루존 E12");
  result.put("save_before_device_ms",System.currentTimeMillis());tap("주차 위치 24시간 기억");locate("주차 위치를 이 기기에 기억했어요.",false,true,false);result.put("save_after_device_ms",System.currentTimeMillis());
  tap("휴대폰으로 가져가기");assertQrAndCapture("standard-qr");tap("QR 크게 보기");assertQrAndCapture("standard-qr-expanded");
  tap("기본 크기");assertQrAndCapture("standard-qr-restored");tap("QR 닫기");
  launchForFont("1.3");tap("주차 위치 기록");check("차량을 안전하게 주차했고, 위치 메모를 확인할게요.",false);
  if(!"B3 블루존 E12".contentEquals(locate("직접 적는 주차 위치 메모, 최대 64자",true,false,false).getText()))throw new AssertionError("Stored memo changed during font/relaunch");
  tap("휴대폰으로 가져가기");assertQrAndCapture("large-font-qr");tap("QR 크게 보기");assertQrAndCapture("large-font-qr-expanded");tap("기본 크기");assertQrAndCapture("large-font-qr-restored");
  result.put("status","passed");passed=true;
 }catch(Throwable error){try{result.put("status","failed_or_incomplete");result.put("error",error.toString());if(ui!=null)result.put("failure_visible_text",allText(ui.getRootInActiveWindow()));if(ui!=null&&out!=null)capture("failure");}catch(Exception ignored){}Bundle b=new Bundle();b.putString("stream",error.toString()+"\n");sendStatus(-1,b);
 }finally{try{shell("settings put system font_scale 1.0");result.put("events",events);try(FileOutputStream f=new FileOutputStream(new File(out,"release-scenario.json"))){f.write(result.toString(2).getBytes("UTF-8"));}}catch(Exception ignored){}Bundle done=new Bundle();done.putString("stream",passed?"RELEASE QR SCENARIO PASSED\n":"RELEASE QR SCENARIO INCOMPLETE\n");finish(passed?-1:0,done);}}
}
